`maplibregl.IControl` インターフェースを実装するカスタムコントロールクラス `BasemapSwitcherControl` を `main.js` に追加する。クリックで展開するドロップダウン形式の UI を構築し、選択された背景地図の visibility を排他的に切り替える。`aria-label` を付与してアクセシビリティにも対応する。

## 主要実装項目

### 1. BasemapSwitcherControl クラス定義 🔵

**信頼性**: 🔵 *architecture.md「BasemapSwitcherControl クラス」・MapLibre GL JS IControl 仕様より*

**実装ファイル**: `main.js`（`map.on('load', ...)` の前に定義）

```javascript
/**
 * 背景地図切り替えコントロール
 * MapLibre GL JS の IControl インターフェースを実装
 */
class BasemapSwitcherControl {
    constructor() {
        // 選択中の背景地図ID（初期値: OSM）
        this._currentBasemap = 'osm';
        // 背景地図の定義リスト
        this._basemaps = [
            { id: 'osm',       layerId: 'osm-layer',       label: 'OSM' },
            { id: 'gsi_std',   layerId: 'gsi-std-layer',   label: '地理院地図' },
            { id: 'gsi_photo', layerId: 'gsi-photo-layer', label: '航空写真' },
        ];
    }

    /**
     * コントロールを地図に追加するときに呼ばれる
     * @param {maplibregl.Map} map
     * @returns {HTMLElement}
     */
    onAdd(map) {
        this._map = map;
        this._container = document.createElement('div');
        this._container.className = 'maplibregl-ctrl maplibregl-ctrl-group basemap-switcher';
        this._buildUI();
        return this._container;
    }

    /**
     * コントロールを地図から除去するときに呼ばれる
     */
    onRemove() {
        this._container.parentNode.removeChild(this._container);
        this._map = undefined;
    }

    /** UI（アイコンボタン + ドロップダウンメニュー）を構築 */
    _buildUI() {
        // アイコンボタン
        const toggleBtn = document.createElement('button');
        toggleBtn.className = 'basemap-switcher-toggle';
        toggleBtn.setAttribute('aria-label', '背景地図を切り替える');
        toggleBtn.textContent = '🗺';
        toggleBtn.addEventListener('click', () => this._toggle());

        // ドロップダウンメニュー
        const menu = document.createElement('div');
        menu.className = 'basemap-switcher-menu';
        menu.hidden = true;

        this._basemaps.forEach(({ id, label }) => {
            const btn = document.createElement('button');
            btn.className = 'basemap-switcher-item' + (id === this._currentBasemap ? ' active' : '');
            btn.textContent = label;
            btn.dataset.id = id;
            btn.addEventListener('click', () => {
                this._switchBasemap(id);
                this._toggle();
            });
            menu.appendChild(btn);
        });

        this._menu = menu;
        this._container.appendChild(toggleBtn);
        this._container.appendChild(menu);
    }

    /** メニューの表示/非表示を切り替え */
    _toggle() {
        this._menu.hidden = !this._menu.hidden;
    }

    /**
     * 背景地図を切り替える
     * @param {string} id - 切り替え先の背景地図ID
     */
    _switchBasemap(id) {
        if (id === this._currentBasemap) return;

        // すべての背景レイヤーを非表示
        this._basemaps.forEach(({ layerId }) => {
            this._map.setLayoutProperty(layerId, 'visibility', 'none');
        });

        // 選択されたレイヤーを表示
        const target = this._basemaps.find((b) => b.id === id);
        if (target) {
            this._map.setLayoutProperty(target.layerId, 'visibility', 'visible');
            this._currentBasemap = id;
        }

        // メニューのアクティブ状態を更新
        this._menu.querySelectorAll('.basemap-switcher-item').forEach((btn) => {
            btn.classList.toggle('active', btn.dataset.id === id);
        });
    }
}
```

## 注意事項

- `BasemapSwitcherControl` クラスは `map.on('load', ...)` の前（グローバルスコープ）に定義すること
- `map.addControl()` の呼び出しは `map.on('load')` コールバック内で行うこと（地図ロード後にコントロールを追加するため）
- `_switchBasemap` 内で `map.setLayoutProperty` を呼び出す際、TASK-0001 で追加したレイヤーID（`'osm-layer'`、`'gsi-std-layer'`、`'gsi-photo-layer'`）を使用すること
