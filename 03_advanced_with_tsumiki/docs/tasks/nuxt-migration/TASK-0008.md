### 1. composables/useShelterLayers.ts 実装 🔵（REQ-030, REQ-031 + main.js:102-363より）

```typescript
import type { Ref } from 'vue'
import type { Map as MaplibreMap, FilterSpecification } from 'maplibre-gl'
import { OpacityControl } from 'maplibre-gl-opacity'
import { SKHB_LAYER_IDS, DISASTER_LABELS } from '~/types'
import type { UseShelterLayersReturn } from '~/types'

export function useShelterLayers(map: Ref<MaplibreMap | null>): UseShelterLayersReturn {
  const getCurrentLayerFilter = (): FilterSpecification | null => {
    const m = map.value
    if (!m) return null
    const visibleLayer = SKHB_LAYER_IDS.find(
      id => m.getLayoutProperty(id, 'visibility') === 'visible'
    )
    if (!visibleLayer) return null
    const match = visibleLayer.match(/skhb-(\d+)-layer/)
    if (!match) return null
    return ['==', ['get', `disaster${match[1]}`], true]
  }

  map.value?.on('load', () => {
    const m = map.value!
    // ベクトルタイルソース追加
    m.addSource('skhb', {
      type: 'vector',
      tiles: [`${location.origin}/skhb/{z}/{x}/{y}.pbf`],
      minzoom: 5,
      maxzoom: 8,
      attribution: '<a href="https://www.gsi.go.jp/">国土地理院</a>',
    })

    const baseLayers: Record<string, string> = {}
    SKHB_LAYER_IDS.forEach((layerId, i) => {
      const disasterNum = i + 1
      m.addLayer({
        id: layerId,
        type: 'circle',
        source: 'skhb',
        'source-layer': 'skhb',
        filter: ['==', ['get', `disaster${disasterNum}`], true],
        paint: {
          'circle-color': '#ff6600',
          'circle-radius': ['interpolate', ['linear'], ['zoom'], 5, 2, 14, 6],
          'circle-opacity': 0.8,
        },
        layout: { visibility: 'none' },
      })
      baseLayers[layerId] = DISASTER_LABELS[disasterNum]
    })

    // OpacityControl（排他制御）を右上に追加
    const opacityControl = new OpacityControl({ baseLayers })
    m.addControl(opacityControl, 'top-right')
  })

  return { getCurrentLayerFilter }
}
```
