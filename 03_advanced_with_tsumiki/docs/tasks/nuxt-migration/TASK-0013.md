## 実装詳細

### 1. MapContainer.vue 実装 🔵

**信頼性**: 🔵 *REQ-011 + アーキテクチャ設計より*

**実装ファイル**: `components/MapContainer.vue`

```vue
<script setup lang="ts">
import { createApp, onMounted } from 'vue'
import { useMap } from '~/composables/useMap'
import { useHazardLayers } from '~/composables/useHazardLayers'
import { useShelterLayers } from '~/composables/useShelterLayers'
import { useGeolocation } from '~/composables/useGeolocation'
import { useRoute } from '~/composables/useRoute'
import { useTerrain } from '~/composables/useTerrain'
import ShelterPopup from '~/components/ShelterPopup.vue'
import { SKHB_LAYER_IDS } from '~/types'
import type { SkhbProperties } from '~/types'

const { map } = useMap('map-container')

onMounted(() => {
  map.value?.on('load', () => {
    // 各composables初期化
    useHazardLayers(map)
    const { getCurrentLayerFilter } = useShelterLayers(map)
    const { userLocation } = useGeolocation(map)
    useRoute(map, userLocation, getCurrentLayerFilter)
    useTerrain(map)

    // 避難場所クリック → ShelterPopup表示
    map.value?.on('click', SKHB_LAYER_IDS, (e) => {
      const feature = e.features?.[0]
      if (!feature) return
      const properties = feature.properties as SkhbProperties
      const container = document.createElement('div')
      const popup = new (window as any).maplibregl.Popup()
        .setLngLat(e.lngLat)
        .setDOMContent(container)
        .addTo(map.value!)
      const app = createApp(ShelterPopup, { properties })
      app.mount(container)
      popup.on('close', () => app.unmount())
    })

    // カーソル変更（REQ-061）
    map.value?.on('mousemove', SKHB_LAYER_IDS, () => {
      if (map.value) map.value.getCanvas().style.cursor = 'pointer'
    })
    map.value?.on('mouseleave', SKHB_LAYER_IDS, () => {
      if (map.value) map.value.getCanvas().style.cursor = ''
    })
  })
})
</script>

<template>
  <div id="map-container" class="map-container" />
</template>

<style scoped>
.map-container {
  width: 100%;
  height: 100vh;
}
</style>
```
