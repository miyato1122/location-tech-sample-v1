### 1. composables/useRoute.ts 実装 🔵（REQ-050, REQ-051, REQ-052 + main.js:371-577より）

```typescript
import type { Ref } from 'vue'
import type { Map as MaplibreMap, FilterSpecification } from 'maplibre-gl'
import distance from '@turf/distance'
import type { UserLocation, SkhbFeatureWithDistance, NearestFeatureResult } from '~/types'

export function getNearestFeature(
  userLocation: [number, number],
  features: GeoJSON.Feature[]
): NearestFeatureResult {
  if (features.length === 0) return null  // EDGE-101修正: 空配列でnull返す

  return features.reduce<SkhbFeatureWithDistance | null>((nearest, feature) => {
    if (feature.geometry.type !== 'Point') return nearest
    const dist = distance(
      [userLocation[0], userLocation[1]],
      feature.geometry.coordinates as [number, number]
    )
    const featureWithDist = { ...feature, properties: { ...feature.properties, dist } } as SkhbFeatureWithDistance
    if (!nearest || dist < nearest.properties.dist) return featureWithDist
    return nearest
  }, null)
}

export function useRoute(
  map: Ref<MaplibreMap | null>,
  userLocation: Ref<UserLocation>,
  getCurrentLayerFilter: () => FilterSpecification | null
): void {
  const clearRoute = () => {
    const source = map.value?.getSource('route-source') as any
    source?.setData({ type: 'FeatureCollection', features: [] })
  }

  map.value?.on('load', () => {
    const m = map.value!
    m.addSource('route-source', {
      type: 'geojson',
      data: { type: 'FeatureCollection', features: [] },
    })
    m.addLayer({
      id: 'route-layer',
      type: 'line',
      source: 'route-source',
      paint: { 'line-color': '#33aaff', 'line-width': 4 },
    })
  })

  map.value?.on('render', () => {
    const m = map.value
    if (!m || m.getZoom() < 7 || !userLocation.value) {
      clearRoute()
      return
    }
    const filter = getCurrentLayerFilter()
    const features = m.querySourceFeatures('skhb', {
      sourceLayer: 'skhb',
      filter: filter ?? undefined,
    })
    const nearest = getNearestFeature(userLocation.value, features)
    if (!nearest) {  // EDGE-101修正: nullチェック
      clearRoute()
      return
    }
    const [lng, lat] = nearest.geometry.coordinates
    const source = m.getSource('route-source') as any
    source?.setData({
      type: 'FeatureCollection',
      features: [{
        type: 'Feature',
        geometry: {
          type: 'LineString',
          coordinates: [userLocation.value, [lng, lat]],
        },
        properties: {},
      }],
    })
  })
}
```
