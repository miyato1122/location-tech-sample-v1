## 実装詳細

### 1. composables/useTerrain.ts 実装 🔵（REQ-070 + main.js:580-602より）

```typescript
import type { Ref } from 'vue'
import type { Map as MaplibreMap } from 'maplibre-gl'
import { useGsiTerrainSource } from 'maplibre-gl-gsi-terrain'

export function useTerrain(map: Ref<MaplibreMap | null>): void {
  map.value?.on('load', async () => {
    const maplibregl = await import('maplibre-gl')
    const m = map.value!

    // GSI標高タイルソース追加
    const { source, terrain } = useGsiTerrainSource(maplibregl.MercatorCoordinate)
    m.addSource('gsi-terrain', source)

    // 陰影図レイヤー追加
    m.addLayer({
      id: 'hillshade',
      type: 'hillshade',
      source: 'gsi-terrain',
      paint: { 'hillshade-exaggeration': 0.2 },
    })

    // TerrainControl追加
    m.addControl(
      new maplibregl.TerrainControl({ source: 'gsi-terrain', exaggeration: 1.5 })
    )
  })
}
```

## 注意事項
