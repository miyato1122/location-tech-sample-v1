## 実装詳細

### 1. composables/useGeolocation.ts 実装 🔵（REQ-040 + main.js:418-425より）

```typescript
import { ref } from 'vue'
import type { Ref } from 'vue'
import type { Map as MaplibreMap } from 'maplibre-gl'
import type { UserLocation, UseGeolocationReturn, GeolocateEventData } from '~/types'

export function useGeolocation(map: Ref<MaplibreMap | null>): UseGeolocationReturn {
  const userLocation = ref<UserLocation>(null)

  map.value?.on('load', async () => {
    const maplibregl = await import('maplibre-gl')
    const geolocationControl = new maplibregl.GeolocateControl({
      trackUserLocation: true,
      fitBoundsOptions: { zoom: 14 },
    })
    map.value!.addControl(geolocationControl)

    geolocationControl.on('geolocate', (e: GeolocateEventData) => {
      userLocation.value = [e.coords.longitude, e.coords.latitude]
    })

    // 追跡停止時に null にリセット
    geolocationControl.on('trackend', () => {
      userLocation.value = null
    })
  })

  return { userLocation }
}
```
