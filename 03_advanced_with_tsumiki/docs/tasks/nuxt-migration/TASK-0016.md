### 1. E2Eテストファイル実装 🔵

**信頼性**: 🔵 *受け入れ基準 TC-010〜TC-080 より*

**実装ファイル**: `tests/e2e/map-flow.spec.ts`

```typescript
import { test, expect } from '@playwright/test'

test.describe('防災マップ E2Eテスト', () => {
  test.beforeEach(async ({ page }) => {
    await page.goto('/')
    // マップ初期化を待つ
    await page.waitForSelector('#map-container canvas')
  })

  test('地図が初期表示される（TC-010-01）', async ({ page }) => {
    const canvas = page.locator('#map-container canvas')
    await expect(canvas).toBeVisible()
  })

  test('OpacityControlが表示される', async ({ page }) => {
    // 左上のハザードマップコントロール
    await expect(page.locator('.maplibregl-ctrl-top-left')).toBeVisible()
    // 右上の避難場所コントロール
    await expect(page.locator('.maplibregl-ctrl-top-right')).toBeVisible()
  })

  test('Geolocationモックでルートが表示される（TC-050-01）', async ({ page, context }) => {
    // Geolocationをモック（東京都庁）
    await context.setGeolocation({ longitude: 139.6917, latitude: 35.6895 })
    await context.grantPermissions(['geolocation'])
    // GeolocateControlをクリック
    await page.click('.maplibregl-ctrl-geolocate')
    // ズーム14でルートが表示されるか確認
    // （ズーム7以上かつ避難場所が表示中の場合のみ）
  })
})
```
