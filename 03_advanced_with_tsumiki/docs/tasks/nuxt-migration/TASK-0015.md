### 1. useRoute.test.ts（EDGE-101修正確認） 🔵

**信頼性**: 🔵 *TC-050-U01, TC-050-U02 + REQ-052より*

**実装ファイル**: `tests/unit/useRoute.test.ts`

```typescript
import { describe, it, expect } from 'vitest'
import { getNearestFeature } from '~/composables/useRoute'

describe('getNearestFeature', () => {
  it('フィーチャーが0件のとき null を返す（EDGE-101修正確認）', () => {
    const result = getNearestFeature([139.7, 35.7], [])
    expect(result).toBeNull()
  })

  it('フィーチャーが1件のとき そのフィーチャーを返す', () => {
    const features = [{
      type: 'Feature',
      geometry: { type: 'Point', coordinates: [139.7, 35.7] },
      properties: { name: 'テスト避難場所', dist: 0 },
    }]
    const result = getNearestFeature([139.7, 35.7], features as any)
    expect(result).not.toBeNull()
    expect(result?.properties.name).toBe('テスト避難場所')
  })

  it('複数フィーチャーのとき最も近いものを返す', () => {
    const features = [
      {
        type: 'Feature',
        geometry: { type: 'Point', coordinates: [140.0, 35.7] },
        properties: { name: '遠い避難場所' },
      },
      {
        type: 'Feature',
        geometry: { type: 'Point', coordinates: [139.71, 35.7] },
        properties: { name: '近い避難場所' },
      },
    ]
    const result = getNearestFeature([139.7, 35.7], features as any)
    expect(result?.properties.name).toBe('近い避難場所')
  })
})
```

### 2. shelterPopup.test.ts（XSS確認） 🔵

**信頼性**: 🔵 *TC-060-S01 + NFR-101より*

**実装ファイル**: `tests/unit/shelterPopup.test.ts`

```typescript
import { describe, it, expect } from 'vitest'
import { mountSuspended } from '@nuxt/test-utils/runtime'
import ShelterPopup from '~/components/ShelterPopup.vue'

const baseProperties = {
  name: 'テスト避難場所',
  address: '東京都新宿区1-1-1',
  remarks: null,
  disaster1: true, disaster2: false, disaster3: false,
  disaster4: true, disaster5: false, disaster6: false,
  disaster7: false, disaster8: false,
}

describe('ShelterPopup', () => {
  it('施設名・住所が表示される', async () => {
    const component = await mountSuspended(ShelterPopup, {
      props: { properties: baseProperties },
    })
    expect(component.text()).toContain('テスト避難場所')
    expect(component.text()).toContain('東京都新宿区1-1-1')
  })

  it('XSS: scriptタグがテキストとして表示される', async () => {
    const xssProperties = {
      ...baseProperties,
      name: '<script>alert("XSS")</script>',
    }
    const component = await mountSuspended(ShelterPopup, {
      props: { properties: xssProperties },
    })
    // scriptタグがHTMLとして解釈されず、テキストとして表示される
    expect(component.html()).not.toContain('<script>')
    expect(component.text()).toContain('<script>alert("XSS")</script>')
  })

  it('remarksがnullのとき備考欄が非表示', async () => {
    const component = await mountSuspended(ShelterPopup, {
      props: { properties: { ...baseProperties, remarks: null } },
    })
    expect(component.find('.shelter-popup__remarks').exists()).toBe(false)
  })
})
```
