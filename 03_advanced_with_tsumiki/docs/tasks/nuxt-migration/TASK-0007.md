## 実装詳細

### 1. composables/useHazardLayers.ts 実装 🔵（REQ-020, REQ-021 + main.js:35-174より）

```typescript
import type { Ref } from 'vue'
import type { Map as MaplibreMap } from 'maplibre-gl'
import { OpacityControl } from 'maplibre-gl-opacity'
import { HAZARD_LAYER_DEFINITIONS } from '~/types'

export function useHazardLayers(map: Ref<MaplibreMap | null>): void {
  map.value?.on('load', () => {
    const m = map.value!
    const overLayers: Record<string, string> = {}

    HAZARD_LAYER_DEFINITIONS.forEach(({ id, label, tileUrl }) => {
      const sourceId = id.replace('-layer', '')
      m.addSource(sourceId, {
        type: 'raster',
        tiles: [tileUrl],
        tileSize: 256,
        attribution: '<a href="https://disaportal.gsi.go.jp/">国土地理院ハザードマップポータルサイト</a>',
      })
      m.addLayer({
        id,
        type: 'raster',
        source: sourceId,
        paint: { 'raster-opacity': 0.7 },
        layout: { visibility: 'none' },
      })
      overLayers[id] = label
    })

    // OpacityControl（maplibre-gl-opacity）を左上に追加
    const opacityControl = new OpacityControl({ overLayers })
    m.addControl(opacityControl, 'top-left')
  })
}
```
